         ```csharp
         var cycles = await _db.Cycles
         ```

         ```csharp
         var now = DateTimeOffset.UtcNow;
         var withStatus = cycles.Select(c => new {
             Cycle = c,
             Status = CycleDtoMapper.ComputeStatus(c)
         }).ToList();

         var filtered = query.CycleView?.ToLowerInvariant() switch
         {
             "current" => withStatus.Where(x => x.Status == "CURRENT").ToList(),
             "upcoming" => withStatus.Where(x => x.Status == "UPCOMING").ToList(),
             "completed" => withStatus.Where(x => x.Status == "COMPLETED").ToList(),
             "draft" => withStatus.Where(x => x.Status == "DRAFT").ToList(),
             "incomplete" => withStatus.Where(x => x.Status != "COMPLETED").ToList(),
             _ => withStatus // "all" or null — no filter
         };
         ```

      4. 对每个结果查询注释字段并映射:
         ```csharp
         var result = new List<CycleDto>();
         foreach (var item in filtered)
         {
             var totalIssues = await _db.CycleIssues
                 .CountAsync(ci => ci.CycleId == item.Cycle.Id && !ci.IsDeleted, ct);
             // ... similar for other annotation fields

             result.Add(CycleDtoMapper.ToDto(
                 item.Cycle, item.Status,
                 totalIssues, completedIssues, cancelledIssues,
                 startedIssues, unstartedIssues, backlogIssues));
         }
         return result;
         ```
