    ```csharp
    public async Task<AnalyticsChartDto> GetWorkspaceWorkItemChartAsync(
        string slug, string? dateFilter, string? startDate, string? endDate, string? projectIds, CancellationToken ct)
    {
        var tenantId = GetTenantId();
        var (dateGte, dateLte) = ParseDateFilter(dateFilter, startDate, endDate);

        var query = _dbContext.Issues
            .Where(i => i.TenantId == tenantId && i.DeletedOnUtc == null);

        if (!string.IsNullOrEmpty(projectIds))
        {
            var pIds = projectIds.Split(',').Select(Guid.Parse).ToList();
            query = query.Where(i => pIds.Contains(i.ProjectId));
        }

        // 确定日期范围起始
        var start = dateGte?.DateTime ?? _dbContext.Issues
            .Where(i => i.TenantId == tenantId && i.DeletedOnUtc == null)
            .MinBy(i => i.CreatedOnUtc)?.CreatedOnUtc.Date ?? DateTime.UtcNow.AddMonths(-12);
        var end = dateLte?.DateTime ?? DateTime.UtcNow;
        // 确保 start 是当月第一天
        start = new DateTime(start.Year, start.Month, 1, 0, 0, 0, DateTimeKind.Utc);

        // 月度聚合
        var monthlyStats = await query
            .Where(i => i.CreatedOnUtc >= start && i.CreatedOnUtc <= end)
            .GroupBy(i => new { i.CreatedOnUtc.Year, i.CreatedOnUtc.Month })
            .Select(g => new
            {
                Year = g.Key.Year,
                Month = g.Key.Month,
                CreatedCount = g.Count(),
                CompletedCount = g.Count(i => i.CompletedAt != null)
            })
            .ToListAsync(ct);

        var statsLookup = monthlyStats.ToDictionary(
            s => $"{s.Year:D4}-{s.Month:D2}",
            s => (s.CreatedCount, s.CompletedCount));

        // 填充所有月份（确保 0 count 的月份也有数据）
        var data = new List<ChartDataPoint>();
        var current = new DateTime(start.Year, start.Month, 1, 0, 0, 0, DateTimeKind.Utc);
        var lastMonth = new DateTime(end.Year, end.Month, 1, 0, 0, 0, DateTimeKind.Utc);

        while (current <= lastMonth)
        {
            var key = current.ToString("yyyy-MM-dd");
            var (createdCount, completedCount) = statsLookup.GetValueOrDefault(
                current.ToString("yyyy-MM"),
                (0, 0));

            data.Add(new ChartDataPoint
            {
                Key = key,
                Name = key,
                Count = createdCount,
                CreatedIssues = createdCount,
                CompletedIssues = completedCount,
            });

            current = current.AddMonths(1);
        }

        return new AnalyticsChartDto
        {
            Data = data,
            Schema = new Dictionary<string, string>
            {
                ["completed_issues"] = "completed_issues",
                ["created_issues"] = "created_issues",
            },
        };
    }
    ```

    **ExportAnalyticsJob.cs** — Hangfire 作业类：
    ```csharp
    public sealed class ExportAnalyticsJob
    {
        private readonly IServiceScopeFactory _scopeFactory;

        public ExportAnalyticsJob(IServiceScopeFactory scopeFactory)
        {
            _scopeFactory = scopeFactory;
        }

        /// <summary>
        /// Hangfire job entry point。在后台生成 CSV。
        /// </summary>
        /// <param name="tenantId">显式传入的租户 ID（作业无 HttpContext）</param>
        /// <param name="slug">工作区 slug</param>
        /// <param name="projectIds">可选项目过滤</param>
        public async Task RunAsync(string tenantId, string slug, string? projectIds)
        {
            using var scope = _scopeFactory.CreateScope();
            var exportService = scope.ServiceProvider.GetRequiredService<IAnalyticsExportService>();
            var csv = await exportService.GenerateCsvAsync(tenantId, slug, projectIds, CancellationToken.None);

            // 目前将 CSV 写入磁盘临时目录（可扩展为发送邮件或提供下载链接）
            var fileName = $"analytics-export-{slug}-{DateTime.UtcNow:yyyyMMdd-HHmmss}.csv";
            var filePath = Path.Combine(Path.GetTempPath(), fileName);
            await File.WriteAllBytesAsync(filePath, csv);
        }
    }
    ```

    **ExportAnalyticsCommandHandler.cs**：
    ```csharp
    public sealed class ExportAnalyticsCommandHandler(
        IJobService jobService,
        IMultiTenantContextAccessor<AppTenantInfo> tenantAccessor)
        : IRequestHandler<ExportAnalyticsCommand, IResult>
    {
        public async Task<IResult> Handle(ExportAnalyticsCommand request, CancellationToken ct)
        {
            var tenantId = tenantAccessor.MultiTenantContext?.TenantInfo?.Id;
            if (string.IsNullOrEmpty(tenantId))
                return Results.Unauthorized();

            // 入队 Hangfire 后台作业
            jobService.Enqueue<ExportAnalyticsJob>(job =>
                job.RunAsync(tenantId, request.Slug, request.ProjectIds));

            return Results.Ok(new
            {
                message = $"Once the export is ready it will be available for download."
            });
        }
    }
    ```
