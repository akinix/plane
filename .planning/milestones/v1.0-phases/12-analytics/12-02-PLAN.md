    5. **GetWorkItemStatsAsync 实现**（按 state group 分组）：
    ```csharp
    public async Task<WorkItemStatsDto> GetWorkItemStatsAsync(
        string slug, string? dateFilter, string? startDate, string? endDate, string? projectIds, CancellationToken ct)
    {
        var tenantId = GetTenantId();
        var (dateGte, dateLte) = ParseDateFilter(dateFilter, startDate, endDate);

        var query = _dbContext.Issues
            .Where(i => i.TenantId == tenantId && i.DeletedOnUtc == null);

        if (!string.IsNullOrEmpty(projectIds))
        {
            var pIds = projectIds.Split(',').Select(Guid.Parse).ToList();
            query = query.Where(i => pIds.Contains(i.ProjectId));
        }

        // 状态分组统计 — 通过 State 导航属性获取 Group
        // EF Core 将生成 LEFT JOIN 到 States 表
        var stats = await query
            .GroupBy(i => i.State != null ? i.State.Group : (StateGroup?)null)
            .Select(g => new { Group = g.Key, Count = g.Count() })
            .ToListAsync(ct);

        var total = await GetFilteredCountAsync(query, dateGte, dateLte, ct);

        // 应用日期过滤到各组
        int GetGroupCount(StateGroup group) =>
            stats.FirstOrDefault(s => s.Group == group)?.Count ?? 0;

        return new WorkItemStatsDto
        {
            TotalWorkItems = new CountValue { Count = total },
            BacklogWorkItems = new CountValue { Count = GetGroupCount(StateGroup.Backlog) },
            UnStartedWorkItems = new CountValue { Count = GetGroupCount(StateGroup.Unstarted) },
            StartedWorkItems = new CountValue { Count = GetGroupCount(StateGroup.Started) },
            CompletedWorkItems = new CountValue { Count = GetGroupCount(StateGroup.Completed) },
            CancelledWorkItems = new CountValue { Count = GetGroupCount(StateGroup.Cancelled) },
        };
    }
    ```

    ```csharp
    public async Task<List<ProjectStatsDto>> GetProjectGroupedStatsAsync(
        string slug, string? dateFilter, string? startDate, string? endDate, string? projectIds, CancellationToken ct)
    {
        var tenantId = GetTenantId();
        var (dateGte, dateLte) = ParseDateFilter(dateFilter, startDate, endDate);

        var query = _dbContext.Issues
            .Where(i => i.TenantId == tenantId && i.DeletedOnUtc == null);

        if (dateGte.HasValue) query = query.Where(i => i.CreatedOnUtc >= dateGte.Value);
        if (dateLte.HasValue) query = query.Where(i => i.CreatedOnUtc <= dateLte.Value);
        if (!string.IsNullOrEmpty(projectIds))
        {
            var pIds = projectIds.Split(',').Select(Guid.Parse).ToList();
            query = query.Where(i => pIds.Contains(i.ProjectId));
        }

        return await query
            .GroupBy(i => new { i.ProjectId, ProjectName = i.Project != null ? i.Project.Name : "" })
            .Select(g => new ProjectStatsDto
            {
                ProjectId = g.Key.ProjectId,
                ProjectName = g.Key.ProjectName,
                BacklogWorkItems = g.Count(i => i.State != null && i.State.Group == StateGroup.Backlog),
                UnStartedWorkItems = g.Count(i => i.State != null && i.State.Group == StateGroup.Unstarted),
                StartedWorkItems = g.Count(i => i.State != null && i.State.Group == StateGroup.Started),
                CompletedWorkItems = g.Count(i => i.State != null && i.State.Group == StateGroup.Completed),
                CancelledWorkItems = g.Count(i => i.State != null && i.State.Group == StateGroup.Cancelled),
            })
            .OrderBy(p => p.ProjectName)
            .ToListAsync(ct);
    }
    ```

    ```csharp
    public async Task<List<AssigneeStatsDto>> GetAssigneeGroupedStatsAsync(

    {

            .Where(ia => ia.TenantId == tenantId && ia.DeletedOnUtc == null && ia.Issue.ProjectId == projectId);

            .GroupBy(ia => new { ia.MemberId, DisplayName = ia.Member != null ? ia.Member.DisplayName : "" })
            .Select(g => new AssigneeStatsDto
            {
                AssigneeId = g.Key.MemberId,
                DisplayName = g.Key.DisplayName,
                BacklogWorkItems = g.Count(ia => ia.Issue.State != null && ia.Issue.State.Group == StateGroup.Backlog),
                UnStartedWorkItems = g.Count(ia => ia.Issue.State != null && ia.Issue.State.Group == StateGroup.Unstarted),
    ```

    ```csharp
    public sealed class GetWorkspaceAnalyticsQueryHandler(
        IAnalyticsQueryService analyticsService,
        WorkItemsDbContext dbContext,
        IMultiTenantContextAccessor<AppTenantInfo> tenantAccessor)
        : IRequestHandler<GetWorkspaceAnalyticsQuery, IResult>
    {
        public async Task<IResult> Handle(GetWorkspaceAnalyticsQuery request, CancellationToken ct)
        {
            var tenantId = tenantAccessor.MultiTenantContext?.TenantInfo?.Id;
            if (string.IsNullOrEmpty(tenantId))
                return Results.Unauthorized();

            // 解析 project_ids
            string? projectIds = request.ProjectIds;

            return request.Tab switch
            {
                "overview" => Results.Ok(await analyticsService.GetWorkspaceOverviewAsync(
                    request.Slug, request.DateFilter, request.StartDate, request.EndDate, projectIds, ct)),
                "work-items" => Results.Ok(await analyticsService.GetWorkItemStatsAsync(
                    request.Slug, request.DateFilter, request.StartDate, request.EndDate, projectIds, ct)),
                _ => Results.BadRequest(new { message = "Invalid tab" }),
            };
        }
    }
    ```

    **Endpoint**:
    ```csharp
    public static class GetWorkspaceAnalyticsEndpoint
    {
        public static RouteHandlerBuilder MapGetWorkspaceAnalyticsEndpoint(this IEndpointRouteBuilder endpoints)
        {
            return endpoints.MapGet("/", async (
                string slug,
                [FromQuery] string? tab,
                [FromQuery(Name = "date_filter")] string? dateFilter,
                [FromQuery(Name = "start_date")] string? startDate,
                [FromQuery(Name = "end_date")] string? endDate,
                [FromQuery(Name = "project_ids")] string? projectIds,
                IMediator mediator,
                CancellationToken ct) =>
            {
                return await mediator.Send(new GetWorkspaceAnalyticsQuery(
                    slug, tab ?? "overview", dateFilter, startDate, endDate, projectIds), ct);
            })
            .WithName("GetWorkspaceAnalytics")
            .Produces<AnalyticsOverviewDto>(StatusCodes.Status200OK)
            .Produces<WorkItemStatsDto>(StatusCodes.Status200OK)
            .Produces(StatusCodes.Status400BadRequest)
            .RequireWorkspaceRole(SlugFromRoute, WorkspaceRole.Admin, WorkspaceRole.Member);
        }
    }
    ```

    ```csharp
    public static class GetWorkspaceStatsEndpoint
    {
        public static RouteHandlerBuilder MapGetWorkspaceStatsEndpoint(this IEndpointRouteBuilder endpoints)
        {
            return endpoints.MapGet("/stats", async (
                string slug,
                [FromQuery] string? type,
                [FromQuery(Name = "date_filter")] string? dateFilter,
                [FromQuery(Name = "start_date")] string? startDate,
                [FromQuery(Name = "end_date")] string? endDate,
                [FromQuery(Name = "project_ids")] string? projectIds,
                IMediator mediator,
                CancellationToken ct) =>
            {
                // 目前只支持 type=work-items
                return Results.Ok(await mediator.Send(
                    new GetWorkspaceStatsQuery(slug, type ?? "work-items", dateFilter, startDate, endDate, projectIds), ct));
            })
            .WithName("GetWorkspaceStats")
            .Produces<List<ProjectStatsDto>>(StatusCodes.Status200OK)
            .RequireWorkspaceRole(SlugFromRoute, WorkspaceRole.Admin, WorkspaceRole.Member);
        }
    }
    ```
