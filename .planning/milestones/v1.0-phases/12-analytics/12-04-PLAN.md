    ```csharp
    public sealed class AnalyticsTestFixture : IDisposable
    {
        private static readonly Guid DefaultTenantId = Guid.Parse("00000000-0000-0000-0000-000000000001");
        private const string DefaultSlug = "test-workspace";

        public WorkItemsDbContext CreateDbContext()
        {
            var accessor = Substitute.For<IMultiTenantContextAccessor<AppTenantInfo>>();
            var tenant = new AppTenantInfo(DefaultTenantId.ToString(), DefaultSlug, "Test Workspace");
            var tenantContext = new MultiTenantContext<AppTenantInfo>(tenant);
            accessor.MultiTenantContext.Returns(tenantContext);

            var options = new DbContextOptionsBuilder<WorkItemsDbContext>()
                .UseInMemoryDatabase($"analytics-{Guid.NewGuid():n}")
                .Options;

            var databaseOptions = Options.Create(new DatabaseOptions
            {
                Provider = "inmemory"
            });
            var environment = Substitute.For<IHostEnvironment>();
            environment.EnvironmentName.Returns("Development");

            return new WorkItemsDbContext(accessor, options, databaseOptions, environment);
        }

        public async Task<WorkItemsDbContext> SeedAsync(WorkItemsDbContext context)
        {
            // 创建测试用的 States
            var projectId = Guid.NewGuid();
            var backlogState = State.Create("Backlog", "#ccc", StateGroup.Backlog, projectId, true);
            var startedState = State.Create("Started", "#3b82f6", StateGroup.Started, projectId, false);
            var completedState = State.Create("Completed", "#22c55e", StateGroup.Completed, projectId, false);
            var cancelledState = State.Create("Cancelled", "#ef4444", StateGroup.Cancelled, projectId, false);

            context.States.AddRange(backlogState, startedState, completedState, cancelledState);
            await context.SaveChangesAsync();

            // 创建不同状态的 Issues
            // backlog issue
            context.Issues.Add(Issue.Create("Backlog Issue", projectId));
            // started issue
            var startedIssue = Issue.Create("Started Issue", projectId);
            // 需要设置 StateId
            context.Issues.Add(startedIssue);
            // completed issue
            var completedIssue = Issue.Create("Completed Issue", projectId);
            context.Issues.Add(completedIssue);

            // 创建 Cycles 和 Modules
            context.Cycles.Add(Cycle.Create("Cycle 1", projectId));
            context.Modules.Add(Module.Create("Module 1", projectId));

            await context.SaveChangesAsync();
            return context;
        }

        public void Dispose()
        {
            GC.SuppressFinalize(this);
        }
    }
    ```

    **重要注意**：Issue 实体使用 private set + factory 方法。测试工厂需要：
    1. 使用 `Issue.Create(name, projectId)` 创建实例
    2. 使用反射设置不可直接赋值的属性（如 `StateId`, `CreatedOnUtc`）
    3. 或添加 internal setter（如果 Issue 允许，参考 Issue.cs 中 `SequenceId` 的 `internal set` 模式）

    实践中，使用 `System.Reflection` 或 `UnsafeAccessor` 设置测试需要的字段。
